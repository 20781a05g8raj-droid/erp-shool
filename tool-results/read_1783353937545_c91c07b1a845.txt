   218→// ===================== Main Component =====================
   219→export function FeesModule() {
   220→  const { user } = useAuthStore();
   221→  const [activeTab, setActiveTab] = useState("overview");
   222→
   223→  const isStudentOrParent = user?.role === "student" || user?.role === "parent";
   224→
   225→  // Shared data
   226→  const [classes, setClasses] = useState<ClassOption[]>([]);
   227→
   228→  // Per-tab data
   229→  const [dashboard, setDashboard] = useState<DashboardData | null>(null);
   230→  const [structures, setStructures] = useState<FeeStructureRow[]>([]);
   231→  const [studentFees, setStudentFees] = useState<StudentFeeRow[]>([]);
   232→
   233→  // Loading flags
   234→  const [loadingClasses, setLoadingClasses] = useState(false);
   235→  const [loadingDashboard, setLoadingDashboard] = useState(false);
   236→  const [loadingStructures, setLoadingStructures] = useState(false);
   237→  const [loadingStudentFees, setLoadingStudentFees] = useState(false);
   238→
   239→  // Filters for student fees tab
   240→  const [filterClassId, setFilterClassId] = useState<string>("all");
   241→  const [filterStatus, setFilterStatus] = useState<string>("all");
   242→  const [searchTerm, setSearchTerm] = useState("");
   243→
   244→  // Dialogs
   245→  const [structureDialogOpen, setStructureDialogOpen] = useState(false);
   246→  const [editingStructure, setEditingStructure] = useState<FeeStructureRow | null>(null);
   247→  const [paymentDialogOpen, setPaymentDialogOpen] = useState(false);
   248→  const [paymentTarget, setPaymentTarget] = useState<StudentFeeRow | null>(null);
   249→  const [detailsDialogOpen, setDetailsDialogOpen] = useState(false);
   250→  const [detailsTarget, setDetailsTarget] = useState<StudentFeeRow | null>(null);
   251→  const [receiptDialogOpen, setReceiptDialogOpen] = useState(false);
   252→  const [receiptData, setReceiptData] = useState<ReceiptPayment | null>(null);
   253→  const [receiptLoading, setReceiptLoading] = useState(false);
   254→
   255→  const fetchClasses = useCallback(async () => {
   256→    setLoadingClasses(true);
   257→    try {
   258→      const data = await apiFetch<{ classes: ClassOption[] }>("/api/classes");
   259→      setClasses(data.classes || []);
   260→    } catch {
   261→      toast.error("Failed to load classes");
   262→    } finally {
   263→      setLoadingClasses(false);
   264→    }
   265→  }, []);
   266→
   267→  const fetchDashboard = useCallback(async () => {
   268→    setLoadingDashboard(true);
   269→    try {
   270→      const data = await apiFetch<DashboardData>("/api/fees/dashboard");
   271→      setDashboard(data);
   272→    } catch {
   273→      toast.error("Failed to load fee dashboard");
   274→    } finally {
   275→      setLoadingDashboard(false);
   276→    }
   277→  }, []);
   278→
   279→  const fetchStructures = useCallback(async () => {
   280→    setLoadingStructures(true);
   281→    try {
   282→      const data = await apiFetch<{ structures: FeeStructureRow[] }>(
   283→        "/api/fees/structures"
   284→      );
   285→      setStructures(data.structures || []);
   286→    } catch {
   287→      toast.error("Failed to load fee structures");
   288→    } finally {
   289→      setLoadingStructures(false);
   290→    }
   291→  }, []);
   292→
   293→  const fetchStudentFees = useCallback(async () => {
   294→    setLoadingStudentFees(true);
   295→    try {
   296→      const params = new URLSearchParams();
   297→      if (!isStudentOrParent && filterClassId !== "all")
   298→        params.set("classId", filterClassId);
   299→      if (filterStatus !== "all") params.set("status", filterStatus);
   300→      if (searchTerm.trim()) params.set("search", searchTerm.trim());
   301→      const qs = params.toString();
   302→      const data = await apiFetch<{ studentFees: StudentFeeRow[] }>(
   303→        `/api/fees${qs ? `?${qs}` : ""}`
   304→      );
   305→      setStudentFees(data.studentFees || []);
   306→    } catch {
   307→      toast.error("Failed to load student fees");
   308→    } finally {
   309→      setLoadingStudentFees(false);
   310→    }
   311→  }, [filterClassId, filterStatus, searchTerm, isStudentOrParent]);
   312→
   313→  // Initial classes fetch (needed by structures + filters)
   314→  useEffect(() => {
   315→    fetchClasses();
   316→  }, [fetchClasses]);
   317→
   318→  // Tab-driven data fetching
   319→  useEffect(() => {
   320→    if (activeTab === "overview") fetchDashboard();
   321→    else if (activeTab === "structures") fetchStructures();
   322→    else if (activeTab === "studentFees") fetchStudentFees();
   323→    // Receipts tab data is sourced from studentFees payments; ensure loaded
   324→    if (activeTab === "receipts" && studentFees.length === 0) fetchStudentFees();
   325→  }, [activeTab, fetchDashboard, fetchStructures, fetchStudentFees, studentFees.length]);
   326→
   327→  // Refresh handler
   328→  const refreshCurrent = useCallback(() => {
   329→    if (activeTab === "overview") fetchDashboard();
   330→    else if (activeTab === "structures") fetchStructures();
   331→    else if (activeTab === "studentFees" || activeTab === "receipts")
   332→      fetchStudentFees();
   333→  }, [activeTab, fetchDashboard, fetchStructures, fetchStudentFees]);
   334→
   335→  // Handlers
   336→  const handleOpenAddStructure = () => {
   337→    setEditingStructure(null);
   338→    setStructureDialogOpen(true);
   339→  };
   340→
   341→  const handleOpenEditStructure = (s: FeeStructureRow) => {
   342→    setEditingStructure(s);
   343→    setStructureDialogOpen(true);
   344→  };
   345→
   346→  const handleDeleteStructure = async (s: FeeStructureRow) => {
   347→    if (
   348→      !confirm(
   349→        `Delete fee structure "${s.name}"? This cannot be undone if no students are assigned.`
   350→      )
   351→    )
   352→      return;
   353→    try {
   354→      await apiFetch(`/api/fees/structures/${s.id}`, { method: "DELETE" });
   355→      toast.success("Fee structure deleted");
   356→      fetchStructures();
   357→    } catch (e) {
   358→      toast.error(e instanceof Error ? e.message : "Delete failed");
   359→    }
   360→  };
   361→
   362→  const handleAssignToClass = async (s: FeeStructureRow) => {
   363→    if (!s.classId) {
   364→      toast.error(
   365→        "This fee structure is not linked to a class. Edit it to set a class first."
   366→      );
   367→      return;
   368→    }
   369→    const cls = classes.find((c) => c.id === s.classId);
   370→    if (
   371→      !confirm(
   372→        `Assign "${s.name}" to all active students in ${cls?.name || "this class"}? Students already assigned will be skipped.`
   373→      )
   374→    )
   375→      return;
   376→    try {
   377→      const res = await apiFetch<{ assigned: number; skipped: number }>(
   378→        "/api/fees",
   379→        {
   380→          method: "POST",
   381→          body: JSON.stringify({
   382→            feeStructureId: s.id,
   383→            classId: s.classId,
   384→            scope: "class",
   385→          }),
   386→        }
   387→      );
   388→      toast.success(
   389→        `Assigned to ${res.assigned} student(s)${
   390→          res.skipped ? `, ${res.skipped} already assigned` : ""
   391→        }.`
   392→      );
   393→      fetchStructures();
   394→    } catch (e) {
   395→      toast.error(e instanceof Error ? e.message : "Assignment failed");
   396→    }
   397→  };
   398→
   399→  const handleOpenPayment = (sf: StudentFeeRow) => {
   400→    setPaymentTarget(sf);
   401→    setPaymentDialogOpen(true);
   402→  };
   403→
   404→  const handleOpenDetails = (sf: StudentFeeRow) => {
   405→    setDetailsTarget(sf);
   406→    setDetailsDialogOpen(true);
   407→  };
   408→
   409→  const handleOpenReceipt = async (paymentId: string) => {
   410→    setReceiptLoading(true);
   411→    setReceiptDialogOpen(true);
   412→    try {
   413→      const data = await apiFetch<{ payment: ReceiptPayment }>(
   414→        `/api/fees/payments/${paymentId}`
   415→      );
   416→      setReceiptData(data.payment);
   417→    } catch (e) {
   418→      toast.error(e instanceof Error ? e.message : "Failed to load receipt");
   419→      setReceiptDialogOpen(false);
   420→    } finally {
   421→      setReceiptLoading(false);
   422→    }
   423→  };
   424→
   425→  const handlePrintReceipt = () => {
   426→    window.print();
   427→  };
   428→
   429→  // Collect all payments across student fees for receipts tab
   430→  const allPayments = useMemo(() => {
   431→    const rows: {
   432→      payment: PaymentRow;
   433→      studentFee: StudentFeeRow;
   434→    }[] = [];
   435→    for (const sf of studentFees) {
   436→      for (const p of sf.payments) {
   437→        rows.push({ payment: p, studentFee: sf });
   438→      }
   439→    }
   440→    rows.sort(
   441→      (a, b) =>
   442→        new Date(b.payment.paymentDate).getTime() -
   443→        new Date(a.payment.paymentDate).getTime()
   444→    );
   445→    return rows;
   446→  }, [studentFees]);
   447→
   448→  return (
   449→    <div>
   450→      <style
   451→        dangerouslySetInnerHTML={{
   452→          __html: `
   453→        @media print {
   454→          html, body { margin: 0 !important; padding: 0 !important; background: #fff !important; }
   455→          body * { visibility: hidden !important; }
   456→          .receipt-print-area, .receipt-print-area * { visibility: visible !important; }
   457→          .receipt-print-area {
   458→            position: fixed !important;
   459→            top: 0 !important; left: 0 !important; right: 0 !important;
   460→            width: 100% !important;
   461→            margin: 0 !important; padding: 0 !important;
   462→            box-shadow: none !important; border: none !important;
   463→            transform: none !important;
   464→            max-height: none !important; overflow: visible !important;
   465→          }
   466→          .receipt-print-area .receipt-paper {
   467→            box-shadow: none !important; border: none !important;
   468→            margin: 0 !important; padding: 16px !important;
   469→            border-radius: 0 !important;
   470→          }
   471→          .no-print { display: none !important; }
   472→        }
   473→        `,
   474→        }}
   475→      />
   476→
   477→      <PageHeader
   478→        title="Fee Management"
   479→        description="Structures, payments, dues tracking, receipts & analytics."
   480→        icon={Wallet}
   481→      />
   482→
   483→      <Tabs value={activeTab} onValueChange={setActiveTab} className="space-y-6">
   484→        <TabsList className="h-9 flex-wrap">
   485→          <TabsTrigger value="overview">Overview</TabsTrigger>
   486→          <TabsTrigger value="structures">Fee Structures</TabsTrigger>
   487→          <TabsTrigger value="studentFees">Student Fees</TabsTrigger>
   488→          <TabsTrigger value="receipts">Receipts</TabsTrigger>
   489→        </TabsList>
   490→
   491→        {/* ===================== OVERVIEW ===================== */}
   492→        <TabsContent value="overview">
   493→          {loadingDashboard || !dashboard ? (
   494→            <OverviewSkeleton />
   495→          ) : (
   496→            <OverviewTab dashboard={dashboard} />
   497→          )}
   498→        </TabsContent>
   499→
   500→        {/* ===================== STRUCTURES ===================== */}
   501→        <TabsContent value="structures">
   502→          <div className="flex justify-end mb-4">
   503→            <Button
   504→              onClick={handleOpenAddStructure}
   505→              className="gradient-primary text-white"
   506→            >
   507→              <Plus className="w-4 h-4 mr-2" /> Add Structure
   508→            </Button>
   509→          </div>
   510→          {loadingStructures ? (
   511→            <StructuresSkeleton />
   512→          ) : structures.length === 0 ? (
   513→            <EmptyState
   514→              icon={Layers}
   515→              title="No fee structures yet"
   516→              description="Create fee structures per class with itemized amounts (tuition, lab, exam, etc.). Then assign to students."
   517→              actionLabel="Add Structure"
   518→              onAction={handleOpenAddStructure}
   519→            />
   520→          ) : (
   521→            <StructuresGrid
   522→              structures={structures}
   523→              onEdit={handleOpenEditStructure}
   524→              onDelete={handleDeleteStructure}
   525→              onAssign={handleAssignToClass}
   526→            />
   527→          )}
   528→        </TabsContent>
   529→
   530→        {/* ===================== STUDENT FEES ===================== */}
   531→        <TabsContent value="studentFees">
   532→          {!isStudentOrParent && (
   533→            <Card className="glass-card p-4 mb-4">
   534→              <div className="flex flex-col md:flex-row gap-3 md:items-end">
   535→                <div className="flex-1">
   536→                  <Label className="mb-1.5">Search Student</Label>
   537→                  <div className="relative">
   538→                    <Search className="w-4 h-4 absolute left-3 top-1/2 -translate-y-1/2 text-muted-foreground" />
   539→                    <Input
   540→                      placeholder="Name or admission no…"
   541→                      value={searchTerm}
   542→                      onChange={(e) => setSearchTerm(e.target.value)}
   543→                      onKeyDown={(e) => e.key === "Enter" && fetchStudentFees()}
   544→                      className="pl-9"
   545→                    />
   546→                  </div>
   547→                </div>
   548→                <div className="w-full md:w-48">
   549→                  <Label className="mb-1.5">Class</Label>
   550→                  <Select
   551→                    value={filterClassId}
   552→                    onValueChange={setFilterClassId}
   553→                  >
   554→                    <SelectTrigger>
   555→                      <SelectValue placeholder="All classes" />
   556→                    </SelectTrigger>
   557→                    <SelectContent>
   558→                      <SelectItem value="all">All classes</SelectItem>
   559→                      {classes.map((c) => (
   560→                        <SelectItem key={c.id} value={c.id}>
   561→                          {c.name}
   562→                        </SelectItem>
   563→                      ))}
   564→                    </SelectContent>
   565→                  </Select>
   566→                </div>
   567→                <div className="w-full md:w-44">
   568→                  <Label className="mb-1.5">Status</Label>
   569→                  <Select value={filterStatus} onValueChange={setFilterStatus}>
   570→                    <SelectTrigger>
   571→                      <SelectValue placeholder="All statuses" />
   572→                    </SelectTrigger>
   573→                    <SelectContent>
   574→                      <SelectItem value="all">All statuses</SelectItem>
   575→                      <SelectItem value="pending">Pending</SelectItem>
   576→                      <SelectItem value="partial">Partial</SelectItem>
   577→                      <SelectItem value="paid">Paid</SelectItem>
   578→                      <SelectItem value="overdue">Overdue</SelectItem>
   579→                    </SelectContent>
   580→                  </Select>
   581→                </div>
   582→                <Button onClick={fetchStudentFees} variant="outline">
   583→                  <Search className="w-4 h-4 mr-2" /> Apply
   584→                </Button>
   585→              </div>
   586→            </Card>
   587→          )}
   588→
   589→          {loadingStudentFees ? (
   590→            <StudentFeesSkeleton />
   591→          ) : studentFees.length === 0 ? (
   592→            <EmptyState
   593→              icon={Users}
   594→              title="No student fee records"
   595→              description={
   596→                isStudentOrParent
   597→                  ? "You don't have any fee records yet."
   598→                  : "Assign fee structures to students from the Fee Structures tab to see them here."
   599→              }
   600→            />
   601→          ) : (
   602→            <StudentFeesTable
   603→              rows={studentFees}
   604→              isStudentOrParent={isStudentOrParent}
   605→              onRecordPayment={handleOpenPayment}
   606→              onViewDetails={handleOpenDetails}
   607→              onOpenReceipt={handleOpenReceipt}
   608→            />
   609→          )}
   610→        </TabsContent>
   611→
   612→        {/* ===================== RECEIPTS ===================== */}
   613→        <TabsContent value="receipts">
   614→          {loadingStudentFees ? (
   615→            <StudentFeesSkeleton />
   616→          ) : allPayments.length === 0 ? (
   617→            <EmptyState
   618→              icon={ReceiptIcon}
   619→              title="No receipts yet"
   620→              description="Once payments are recorded, receipts will appear here for printing."
   621→            />
   622→          ) : (
   623→            <ReceiptsTable
   624→              rows={allPayments}
   625→              onOpenReceipt={handleOpenReceipt}
   626→            />
   627→          )}
   628→        </TabsContent>
   629→      </Tabs>
   630→
   631→      {/* ===================== Dialogs ===================== */}

   977→// ===================== Structure Dialog (Add / Edit) =====================
   978→function StructureDialog({
   979→  open,
   980→  onOpenChange,
   981→  classes,
   982→  editing,
   983→  loadingClasses,
   984→  onSaved,
   985→}: {
   986→  open: boolean;
   987→  onOpenChange: (v: boolean) => void;
   988→  classes: ClassOption[];
   989→  editing: FeeStructureRow | null;
   990→  loadingClasses: boolean;
   991→  onSaved: () => void;
   992→}) {
   993→  const [name, setName] = useState("");
   994→  const [classId, setClassId] = useState<string>("");
   995→  const [term, setTerm] = useState("Annual");
   996→  const [dueDate, setDueDate] = useState("");
   997→  const [items, setItems] = useState<FeeItemInput[]>([
   998→    { name: "Tuition Fee", amount: 0 },
   999→  ]);
  1000→  const [saving, setSaving] = useState(false);
  1001→
  1002→  useEffect(() => {
  1003→    if (open) {
  1004→      if (editing) {
  1005→        setName(editing.name);
  1006→        setClassId(editing.classId || "");
  1007→        setTerm(editing.term);
  1008→        setDueDate(editing.dueDate || "");
  1009→        setItems(
  1010→          editing.items.length > 0
  1011→            ? editing.items.map((it) => ({ name: it.name, amount: it.amount }))
  1012→            : [{ name: "Tuition Fee", amount: 0 }]
  1013→        );
  1014→      } else {
  1015→        setName("");
  1016→        setClassId("");
  1017→        setTerm("Annual");
  1018→        setDueDate("");
  1019→        setItems([{ name: "Tuition Fee", amount: 0 }]);
  1020→      }
  1021→    }
  1022→  }, [open, editing]);
  1023→
  1024→  const totalAmount = useMemo(
  1025→    () => items.reduce((s, it) => s + (Number(it.amount) || 0), 0),
  1026→    [items]
  1027→  );
  1028→
  1029→  const handleAddItem = () =>
  1030→    setItems((prev) => [...prev, { name: "", amount: 0 }]);
  1031→
  1032→  const handleRemoveItem = (idx: number) =>
  1033→    setItems((prev) => prev.filter((_, i) => i !== idx));
  1034→
  1035→  const handleItemChange = (idx: number, field: keyof FeeItemInput, value: string) => {
  1036→    setItems((prev) =>
  1037→      prev.map((it, i) =>
  1038→        i === idx
  1039→          ? field === "amount"
  1040→            ? { ...it, amount: Number(value) || 0 }
  1041→            : { ...it, name: value }
  1042→          : it
  1043→      )
  1044→    );
  1045→  };
  1046→
  1047→  const handleSave = async () => {
  1048→    if (!name.trim() || !term.trim()) {
  1049→      toast.error("Name and term are required");
  1050→      return;
  1051→    }
  1052→    const cleanItems = items
  1053→      .filter((it) => it.name.trim() && Number(it.amount) > 0)
  1054→      .map((it) => ({ name: it.name.trim(), amount: Number(it.amount) }));
  1055→
  1056→    setSaving(true);
  1057→    try {
  1058→      const payload = {
  1059→        name: name.trim(),
  1060→        classId: classId || null,
  1061→        term: term.trim(),
  1062→        dueDate: dueDate || null,
  1063→        items: cleanItems,
  1064→      };
  1065→      if (editing) {
  1066→        await apiFetch(`/api/fees/structures/${editing.id}`, {
  1067→          method: "PUT",
  1068→          body: JSON.stringify(payload),
  1069→        });
  1070→        toast.success("Fee structure updated");
  1071→      } else {
  1072→        await apiFetch("/api/fees/structures", {
  1073→          method: "POST",
  1074→          body: JSON.stringify(payload),
  1075→        });
  1076→        toast.success("Fee structure created");
  1077→      }
  1078→      onSaved();
  1079→    } catch (e) {
  1080→      toast.error(e instanceof Error ? e.message : "Save failed");
  1081→    } finally {
  1082→      setSaving(false);
  1083→    }
  1084→  };
  1085→
  1086→  return (
  1087→    <Dialog open={open} onOpenChange={onOpenChange}>
  1088→      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto">
  1089→        <DialogHeader>
  1090→          <DialogTitle>
  1091→            {editing ? "Edit Fee Structure" : "Add Fee Structure"}
  1092→          </DialogTitle>
  1093→          <DialogDescription>
  1094→            Define fee components (tuition, lab, exam, etc.). The total is auto-calculated.
  1095→          </DialogDescription>
  1096→        </DialogHeader>
  1097→
  1098→        <div className="space-y-4 py-2">
  1099→          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
  1100→            <div className="space-y-1.5">
  1101→              <Label>Structure Name *</Label>
  1102→              <Input
  1103→                value={name}
  1104→                onChange={(e) => setName(e.target.value)}
  1105→                placeholder="e.g. Class 10 - Annual Fee 2024-25"
  1106→              />
  1107→            </div>
  1108→            <div className="space-y-1.5">
  1109→              <Label>Class</Label>
  1110→              <Select value={classId} onValueChange={setClassId}>
  1111→                <SelectTrigger>
  1112→                  <SelectValue placeholder={loadingClasses ? "Loading…" : "Select class"} />
  1113→                </SelectTrigger>
  1114→                <SelectContent>
  1115→                  {classes.map((c) => (
  1116→                    <SelectItem key={c.id} value={c.id}>
  1117→                      {c.name}
  1118→                    </SelectItem>
  1119→                  ))}
  1120→                </SelectContent>
  1121→              </Select>
  1122→            </div>
  1123→            <div className="space-y-1.5">
  1124→              <Label>Term *</Label>
  1125→              <Select value={term} onValueChange={setTerm}>
  1126→                <SelectTrigger>
  1127→                  <SelectValue />
  1128→                </SelectTrigger>
  1129→                <SelectContent>
  1130→                  <SelectItem value="Annual">Annual</SelectItem>
  1131→                  <SelectItem value="Term 1">Term 1</SelectItem>
  1132→                  <SelectItem value="Term 2">Term 2</SelectItem>
  1133→                  <SelectItem value="Term 3">Term 3</SelectItem>
  1134→                  <SelectItem value="Quarterly">Quarterly</SelectItem>
  1135→                  <SelectItem value="Monthly">Monthly</SelectItem>
  1136→                  <SelectItem value="One-time">One-time</SelectItem>
  1137→                </SelectContent>
  1138→              </Select>
  1139→            </div>
  1140→            <div className="space-y-1.5">
  1141→              <Label>Due Date</Label>
  1142→              <Input
  1143→                type="date"
  1144→                value={dueDate}
  1145→                onChange={(e) => setDueDate(e.target.value)}
  1146→              />
  1147→            </div>
  1148→          </div>
  1149→
  1150→          <div className="space-y-2">
  1151→            <div className="flex items-center justify-between">
  1152→              <Label>Fee Items</Label>
  1153→              <Button type="button" size="sm" variant="outline" onClick={handleAddItem}>
  1154→                <Plus className="w-3.5 h-3.5 mr-1" /> Add Item
  1155→              </Button>
  1156→            </div>
  1157→            <div className="space-y-2">
  1158→              {items.map((it, idx) => (
  1159→                <div key={idx} className="flex gap-2 items-center">
  1160→                  <Input
  1161→                    value={it.name}
  1162→                    onChange={(e) => handleItemChange(idx, "name", e.target.value)}
  1163→                    placeholder="Item name (e.g. Tuition Fee)"
  1164→                    className="flex-1"
  1165→                  />
  1166→                  <Input
  1167→                    type="number"
  1168→                    min={0}
  1169→                    value={it.amount || ""}
  1170→                    onChange={(e) => handleItemChange(idx, "amount", e.target.value)}
  1171→                    placeholder="Amount"
  1172→                    className="w-32"
  1173→                  />
  1174→                  <Button
  1175→                    type="button"
  1176→                    size="icon"
  1177→                    variant="ghost"
  1178→                    onClick={() => handleRemoveItem(idx)}
  1179→                    disabled={items.length === 1}
  1180→                    className="text-red-500 hover:text-red-600"
  1181→                  >
  1182→                    <Trash2 className="w-4 h-4" />
  1183→                  </Button>
  1184→                </div>
  1185→              ))}
  1186→            </div>
  1187→            <div className="flex justify-end pt-2 border-t">

  1193→              </div>
  1194→            </div>
  1195→          </div>
  1196→        </div>
  1197→
  1198→        <DialogFooter>
