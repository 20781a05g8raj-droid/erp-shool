   181→// ==================== Component ====================
   182→
   183→export function StudentsModule() {
   184→  const [students, setStudents] = useState<StudentListItem[]>([]);
   185→  const [classes, setClasses] = useState<ClassOption[]>([]);
   186→  const [loading, setLoading] = useState(true);
   187→  const [submitting, setSubmitting] = useState(false);
   188→
   189→  // Filters
   190→  const [search, setSearch] = useState("");
   191→  const [classFilter, setClassFilter] = useState("all");
   192→  const [statusFilter, setStatusFilter] = useState("all");
   193→  const [page, setPage] = useState(1);
   194→
   195→  // Dialog (Add/Edit)
   196→  const [dialogOpen, setDialogOpen] = useState(false);
   197→  const [editingId, setEditingId] = useState<string | null>(null);
   198→  const [form, setForm] = useState<StudentFormState>(EMPTY_FORM);
   199→
   200→  // Detail sheet
   201→  const [detailOpen, setDetailOpen] = useState(false);
   202→  const [detailLoading, setDetailLoading] = useState(false);
   203→  const [detail, setDetail] = useState<StudentDetail | null>(null);
   204→
   205→  // Delete
   206→  const [deleteTarget, setDeleteTarget] = useState<StudentListItem | null>(null);
   207→  const [deleting, setDeleting] = useState(false);
   208→
   209→  // ---------- Data fetching ----------
   210→  const fetchStudents = useCallback(async () => {
   211→    try {
   212→      setLoading(true);
   213→      const params = new URLSearchParams();
   214→      if (search.trim()) params.set("search", search.trim());
   215→      if (classFilter !== "all") params.set("classId", classFilter);
   216→      if (statusFilter !== "all") params.set("status", statusFilter);
   217→      const query = params.toString() ? `?${params.toString()}` : "";
   218→      const data = await apiFetch<StudentListItem[]>(`/api/students${query}`);
   219→      setStudents(data);
   220→    } catch (err) {
   221→      toast.error(err instanceof Error ? err.message : "Failed to load students");
   222→    } finally {
   223→      setLoading(false);
   224→    }
   225→  }, [search, classFilter, statusFilter]);
   226→
   227→  const fetchClasses = useCallback(async () => {
   228→    try {
   229→      const data = await apiFetch<ClassOption[]>(`/api/classes`);
   230→      setClasses(data);
   231→    } catch {
   232→      // silent — selects will just be empty
   233→    }
   234→  }, []);
   235→
   236→  useEffect(() => {
   237→    void fetchStudents();
   238→  }, [fetchStudents]);
   239→
   240→  useEffect(() => {
   241→    void fetchClasses();
   242→  }, [fetchClasses]);
   243→
   244→  // Debounced search trigger
   245→  useEffect(() => {
   246→    const t = setTimeout(() => {
   247→      setPage(1);
   248→    }, 300);
   249→    return () => clearTimeout(t);
   250→  }, [search]);
   251→
   252→  // ---------- Derived stats ----------
   253→  const stats = useMemo(() => {
   254→    const total = students.length;
   255→    const active = students.filter((s) => s.status === "active").length;
   256→    const now = new Date();
   257→    const newThisMonth = students.filter((s) => {
   258→      if (!s.admissionDate) return false;
   259→      const d = new Date(s.admissionDate);
   260→      return (
   261→        d.getMonth() === now.getMonth() && d.getFullYear() === now.getFullYear()
   262→      );
   263→    }).length;
   264→    const inactive = students.filter(
   265→      (s) => s.status === "transferred" || s.status === "alumni"
   266→    ).length;
   267→    return { total, active, newThisMonth, inactive };
   268→  }, [students]);
   269→
   270→  // ---------- Pagination ----------
   271→  const totalPages = Math.max(1, Math.ceil(students.length / PAGE_SIZE));
   272→  const currentPage = Math.min(page, totalPages);
   273→  const paged = students.slice(
   274→    (currentPage - 1) * PAGE_SIZE,
   275→    currentPage * PAGE_SIZE
   276→  );
   277→
   278→  // ---------- Dialog helpers ----------
   279→  const openAddDialog = () => {
   280→    setEditingId(null);
   281→    setForm(EMPTY_FORM);
   282→    setDialogOpen(true);
   283→  };
   284→
   285→  const openEditDialog = (student: StudentListItem) => {
   286→    setEditingId(student.id);
   287→    setForm({
   288→      admissionNumber: student.admissionNumber,
   289→      rollNumber: student.rollNumber || "",
   290→      firstName: student.firstName,
   291→      lastName: student.lastName,
   292→      email: student.email || "",
   293→      phone: student.phone || "",
   294→      dob: "",
   295→      gender: student.gender || "",
   296→      bloodGroup: "",
   297→      address: "",
   298→      photo: student.photo || "",
   299→      classId: student.classId || "",
   300→      sectionId: student.sectionId || "",
   301→      status: student.status,
   302→      fatherName: student.fatherName || "",
   303→      motherName: "",
   304→      parentPhone: student.parentPhone || "",
   305→      parentEmail: "",
   306→      admissionDate: student.admissionDate || "",
   307→    });
   308→    setDialogOpen(true);
   309→  };
   310→
   311→  // When editing an existing student, fetch full detail to prefill all fields
   312→  useEffect(() => {
   313→    if (editingId && dialogOpen) {
   314→      void (async () => {
   315→        try {
   316→          const d = await apiFetch<StudentDetail>(`/api/students/${editingId}`);
   317→          setForm((prev) => ({
   318→            ...prev,
   319→            dob: d.dob || "",
   320→            gender: d.gender || prev.gender,
   321→            bloodGroup: d.bloodGroup || "",
   322→            address: d.address || "",
   323→            motherName: d.motherName || "",
   324→            parentEmail: d.parentEmail || "",
   325→            admissionDate: d.admissionDate || prev.admissionDate,
   326→            photo: d.photo || prev.photo,
   327→          }));
   328→        } catch {
   329→          // ignore — basic fields already prefilled
   330→        }
   331→      })();
   332→    }
   333→  }, [editingId, dialogOpen]);
   334→
   335→  const updateField = <K extends keyof StudentFormState>(
   336→    key: K,
   337→    value: StudentFormState[K]
   338→  ) => {
   339→    setForm((prev) => ({ ...prev, [key]: value }));
   340→  };
   341→
   342→  const handleSubmit = async () => {
   343→    if (!form.firstName.trim() || !form.lastName.trim()) {
   344→      toast.error("First name and last name are required");
   345→      return;
   346→    }
   347→    setSubmitting(true);
   348→    try {
   349→      const payload = { ...form };
   350→      if (editingId) {
   351→        await apiFetch(`/api/students/${editingId}`, {
   352→          method: "PUT",
   353→          body: JSON.stringify(payload),
   354→        });
   355→        toast.success("Student updated successfully");
   356→      } else {
   357→        await apiFetch(`/api/students`, {
   358→          method: "POST",
   359→          body: JSON.stringify(payload),
   360→        });
   361→        toast.success("Student admitted successfully");
   362→      }
   363→      setDialogOpen(false);
   364→      setEditingId(null);
   365→      setForm(EMPTY_FORM);
   366→      await fetchStudents();
   367→    } catch (err) {
   368→      toast.error(err instanceof Error ? err.message : "Failed to save student");
   369→    } finally {
   370→      setSubmitting(false);
   371→    }
   372→  };
   373→
   374→  // ---------- Detail ----------
   375→  const openDetail = async (student: StudentListItem) => {
   376→    setDetailOpen(true);
   377→    setDetail(null);
   378→    setDetailLoading(true);
   379→    try {
   380→      const data = await apiFetch<StudentDetail>(`/api/students/${student.id}`);
   381→      setDetail(data);
   382→    } catch (err) {
   383→      toast.error(err instanceof Error ? err.message : "Failed to load student");
   384→      setDetailOpen(false);
   385→    } finally {
   386→      setDetailLoading(false);
   387→    }
   388→  };
   389→
   390→  // ---------- Delete ----------
   391→  const confirmDelete = async () => {
   392→    if (!deleteTarget) return;
   393→    setDeleting(true);
   394→    try {
   395→      await apiFetch(`/api/students/${deleteTarget.id}`, { method: "DELETE" });
   396→      toast.success("Student deleted successfully");
   397→      setDeleteTarget(null);
   398→      if (detail?.id === deleteTarget.id) {
   399→        setDetailOpen(false);
   400→        setDetail(null);
   401→      }
   402→      await fetchStudents();
   403→    } catch (err) {
   404→      toast.error(err instanceof Error ? err.message : "Failed to delete student");
   405→    } finally {
   406→      setDeleting(false);
   407→    }
   408→  };
   409→
   410→  // ---------- Sections for form ----------
   411→  const selectedClass = classes.find((c) => c.id === form.classId);
   412→  const availableSections = selectedClass?.sections || [];
   413→
   414→  // ---------- Render ----------
   415→  return (
   416→    <div className="space-y-6">
   417→      <PageHeader
   418→        title="Students"
   419→        description="Manage student admissions, profiles, and class assignments."
   420→        icon={Users}
   421→        actionLabel="Add Student"
   422→        onAction={openAddDialog}
   423→      />
   424→
   425→      {/* Stats row */}
   426→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   427→        <StatsCard title="Total Students" value={stats.total} icon={Users} delay={0} />
   428→        <StatsCard title="Active" value={stats.active} icon={GraduationCap} delay={0.05} color="text-emerald-500" />
   429→        <StatsCard title="New This Month" value={stats.newThisMonth} icon={UserPlus} delay={0.1} color="text-blue-500" />
   430→        <StatsCard title="Alumni / Transferred" value={stats.inactive} icon={ClipboardCheck} delay={0.15} color="text-amber-500" />
   431→      </div>
   432→
   433→      {/* Filter bar */}
   434→      <Card className="glass-card p-4">
   435→        <div className="flex flex-col md:flex-row gap-3">
   436→          <div className="relative flex-1">
   437→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   438→            <Input
   439→              placeholder="Search by name, admission no, phone, email..."
   440→              value={search}
   441→              onChange={(e) => setSearch(e.target.value)}
   442→              className="pl-9"
   443→            />
   444→          </div>
   445→          <Select value={classFilter} onValueChange={setClassFilter}>
   446→            <SelectTrigger className="w-full md:w-[180px]">
   447→              <SelectValue placeholder="All Classes" />
   448→            </SelectTrigger>
   449→            <SelectContent>
   450→              <SelectItem value="all">All Classes</SelectItem>
   451→              {classes.map((c) => (
   452→                <SelectItem key={c.id} value={c.id}>
   453→                  {c.name}
   454→                </SelectItem>
   455→              ))}
   456→            </SelectContent>
   457→          </Select>
   458→          <Select value={statusFilter} onValueChange={setStatusFilter}>
   459→            <SelectTrigger className="w-full md:w-[160px]">
   460→              <SelectValue placeholder="All Status" />
   461→            </SelectTrigger>
   462→            <SelectContent>
   463→              <SelectItem value="all">All Status</SelectItem>
   464→              {STATUS_OPTIONS.map((s) => (
   465→                <SelectItem key={s.value} value={s.value}>
   466→                  {s.label}
   467→                </SelectItem>
   468→              ))}
   469→            </SelectContent>
   470→          </Select>
   471→        </div>
   472→      </Card>
   473→
   474→      {/* Table or Empty / Loading */}
   475→      {loading ? (
   476→        <Card className="glass-card p-6 space-y-3">
   477→          {[...Array(6)].map((_, i) => (
   478→            <div key={i} className="h-12 rounded-md bg-muted/40 animate-pulse" />
   479→          ))}
   480→        </Card>
   481→      ) : students.length === 0 ? (
   482→        <Card className="glass-card">
   483→          <EmptyState
   484→            icon={Users}
   485→            title="No students yet"
   486→            description="Add your first student to start managing admissions, profiles, and class assignments."
   487→            actionLabel="Add Student"
   488→            onAction={openAddDialog}
   489→          />
   490→        </Card>
   491→      ) : (
   492→        <Card className="glass-card overflow-hidden">
   493→          <div className="max-h-[560px] overflow-y-auto">
   494→            <Table>
   495→              <TableHeader className="sticky top-0 z-10 bg-background/95 backdrop-blur">
   496→                <TableRow>
   497→                  <TableHead className="min-w-[220px]">Student</TableHead>
   498→                  <TableHead>Class / Section</TableHead>
   499→                  <TableHead>Contact</TableHead>
   500→                  <TableHead>Parent / Guardian</TableHead>
   501→                  <TableHead>Status</TableHead>
   502→                  <TableHead className="text-right">Actions</TableHead>
   503→                </TableRow>
   504→              </TableHeader>
   505→              <TableBody>
   506→                {paged.map((student, idx) => (
   507→                  <motion.tr
   508→                    key={student.id}
   509→                    initial={{ opacity: 0, y: 6 }}
   510→                    animate={{ opacity: 1, y: 0 }}
   511→                    transition={{ delay: Math.min(idx * 0.02, 0.2), duration: 0.25 }}
   512→                    className="hover:bg-muted/50 border-b transition-colors"
   513→                  >
   514→                    <TableCell className="py-3">
   515→                      <div className="flex items-center gap-3">
   516→                        <Avatar className="w-9 h-9">
   517→                          {student.photo ? (
   518→                            <AvatarImage src={student.photo} alt={`${student.firstName} ${student.lastName}`} />
   519→                          ) : null}
   520→                          <AvatarFallback className="bg-primary/10 text-primary text-xs font-semibold">
   521→                            {getInitials(`${student.firstName} ${student.lastName}`)}
   522→                          </AvatarFallback>
   523→                        </Avatar>
   524→                        <div className="min-w-0">
   525→                          <div className="font-medium text-sm truncate">
   526→                            {student.firstName} {student.lastName}
   527→                          </div>
   528→                          <div className="text-xs text-muted-foreground">
   529→                            {student.admissionNumber}
   530→                            {student.rollNumber ? ` • #${student.rollNumber}` : ""}
   531→                          </div>
   532→                        </div>
   533→                      </div>
   534→                    </TableCell>
   535→                    <TableCell>
   536→                      <div className="text-sm">
   537→                        <div className="font-medium">{student.class?.name || "—"}</div>
   538→                        <div className="text-xs text-muted-foreground">
   539→                          {student.section ? `Sec ${student.section.name}` : "—"}
   540→                        </div>
   541→                      </div>
   542→                    </TableCell>
   543→                    <TableCell>
   544→                      <div className="text-sm space-y-0.5">
   545→                        <div className="flex items-center gap-1.5 text-xs">
   546→                          <Phone className="w-3 h-3 text-muted-foreground" />
   547→                          <span className="truncate">{student.phone || "—"}</span>
   548→                        </div>
   549→                        <div className="flex items-center gap-1.5 text-xs text-muted-foreground">
   550→                          <Mail className="w-3 h-3" />
   551→                          <span className="truncate max-w-[180px]">{student.email || "—"}</span>
   552→                        </div>
   553→                      </div>
   554→                    </TableCell>
   555→                    <TableCell>
   556→                      <div className="text-sm space-y-0.5">
   557→                        <div className="font-medium">{student.fatherName || "—"}</div>
   558→                        <div className="text-xs text-muted-foreground">{student.parentPhone || "—"}</div>
   559→                      </div>
   560→                    </TableCell>
   561→                    <TableCell>
   562→                      <Badge
   563→                        variant="outline"
   564→                        className={STUDENT_STATUS_STYLES[student.status] || "bg-muted text-muted-foreground"}
   565→                      >
   566→                        {student.status.charAt(0).toUpperCase() + student.status.slice(1)}
   567→                      </Badge>
   568→                    </TableCell>
   569→                    <TableCell className="text-right">
   570→                      <DropdownMenu>
   571→                        <DropdownMenuTrigger asChild>
   572→                          <Button variant="ghost" size="icon" className="h-8 w-8">
   573→                            <MoreHorizontal className="w-4 h-4" />
   574→                            <span className="sr-only">Open menu</span>
   575→                          </Button>
   576→                        </DropdownMenuTrigger>
   577→                        <DropdownMenuContent align="end">
   578→                          <DropdownMenuItem onClick={() => void openDetail(student)}>
   579→                            <Eye className="w-4 h-4 mr-2" />
   580→                            View Profile
