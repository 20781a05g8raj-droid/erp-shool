   518→CREATE OR REPLACE FUNCTION update_updated_at()
   519→RETURNS TRIGGER AS $$
   520→BEGIN
   521→  NEW.updated_at = now();
   522→  RETURN NEW;
   523→END;

   540→RETURNS TRIGGER AS $$
   541→BEGIN
   542→  INSERT INTO public.profiles (id, email, name)
   543→  VALUES (
   544→    NEW.id,
   545→    NEW.email,
   546→    COALESCE(NEW.raw_user_meta_data->>'name', split_part(NEW.email, '@', 1))
   547→  );
   548→  RETURN NEW;
   549→END;
   550→$$ LANGUAGE plpgsql SECURITY DEFINER;
   551→
   552→CREATE TRIGGER on_auth_user_created
   553→  AFTER INSERT ON auth.users
   554→  FOR EACH ROW EXECUTE FUNCTION handle_new_user();
